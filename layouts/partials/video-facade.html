{{- /* Click-to-load video facade. Params (dict): id, title, poster. `poster` is a path under assets/, not static/: Hugo can then read the image's real dimensions for the width/height attributes, fingerprint it, and — the point of the guard below — fail the build if it is missing, rather than shipping a broken player. The poster is hosted here rather than fetched from i.ytimg.com so that the page contacts no third party before the visitor chooses to watch. See assets/js/video-facade.js. */ -}} {{- $id := .id -}} {{- $title := .title | default "Video" -}} {{- if not .poster -}} {{- errorf "video-facade: video %q has no poster. Add one under assets/images/video/ and pass it as `poster`." $id -}} {{- end -}} {{- $poster := resources.Get .poster -}} {{- if not $poster -}} {{- errorf "video-facade: poster %q not found under assets/ (video %q)" .poster $id -}} {{- end -}} {{- $poster = $poster | fingerprint -}}
{{- /* No JavaScript: the visitor goes to YouTube themselves, on the nocookie domain, which is their decision rather than the page's. */ -}}
