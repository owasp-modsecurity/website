{{- /* Returns the "articles" or "news" page set for the blog section. "News" is defined as: blog-section pages carrying the Modsecurity-News tag. "Articles" is defined as: blog-section pages NOT carrying that tag. This is the single place in the codebase that names the tag, and the single place that scopes the split to the blog section (rather than the whole site). Params (dict): context - a page in the current site (used to reach .Site) mode - "articles" (default) or "news" */ -}} {{- $newsTag := site.Params.news_tag | default "Modsecurity-News" -}} {{- $mode := .mode | default "articles" -}} {{- $blogSection := .context.Site.GetPage "blog" -}} {{- $result := slice -}} {{- with $blogSection -}} {{- range .RegularPages -}} {{- /* Case-insensitive on purpose: `in` is exact, so an editor writing "ModSecurity-News" (the capitalisation the site's own prose uses) instead of "Modsecurity-News" would empty /news/ and silently push every news post back into /blog/, with a green build. */ -}} {{- $isNews := false -}} {{- range .Params.tags -}} {{- if eq (lower .) (lower $newsTag) -}}{{- $isNews = true -}}{{- end -}} {{- end -}} {{- if eq $mode "news" -}} {{- if $isNews -}}{{- $result = $result | append . -}}{{- end -}} {{- else -}} {{- if not $isNews -}}{{- $result = $result | append . -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
